# 一、装饰器（Decorator）

## 1. 为什么存在

**核心问题：** 如何在不修改原函数的情况下，给函数添加额外功能？

```python
# 没有装饰器时，你得这样做：
def call_llm(prompt):
    print("start logging")  # 每个函数都要复制这段
    result = api.call(prompt)
    print("end logging")    # 违反 DRY 原则
    return result

def search_web(query):
    print("start logging")  # 重复！
    result = web.search(query)
    print("end logging")    # 重复！
    return result
```

装饰器解决的是**横切关注点（Cross-cutting Concerns）**：日志、重试、缓存、权限验证、计时……

---

## 2. Python 底层原理

**装饰器本质：函数是一等公民 + 闭包**

```python
# Python 中函数可以：
# 1. 作为参数传递
# 2. 作为返回值
# 3. 在函数内部定义（闭包）

def my_decorator(func):          # 接收函数作为参数
    def wrapper(*args, **kwargs): # 内部定义新函数（闭包）
        print("before")
        result = func(*args, **kwargs)  # 调用原函数
        print("after")
        return result
    return wrapper               # 返回新函数

# @语法糖 等价于：
# say_hello = my_decorator(say_hello)
@my_decorator
def say_hello(name):
    print(f"Hello, {name}")

# 验证底层机制
say_hello("Alice")
# before
# Hello, Alice
# after
```

**`\*args, **kwargs` 的重要性：\*\* wrapper 必须能接收任意参数，才能透明代理原函数。

---

## 3. 语法和核心用法

### 3.1 基础装饰器

```python
import time
import functools

def timer(func):
    @functools.wraps(func)  # 保留原函数元信息！重要！
    def wrapper(*args, **kwargs):
        start = time.perf_counter()
        result = func(*args, **kwargs)
        elapsed = time.perf_counter() - start
        print(f"[{func.__name__}] 耗时: {elapsed:.4f}s")
        return result
    return wrapper

@timer
def slow_function():
    time.sleep(0.1)
    return "done"

slow_function()
# [slow_function] 耗时: 0.1002s
```

> ⚠️ **`@functools.wraps(func)` 为什么必须加？**
>
> 没有它，`slow_function.__name__` 会变成 `"wrapper"`，`__doc__` 也会丢失。调试和文档生成都会出问题。

### 3.2 带参数的装饰器（三层嵌套）

```python
def retry(max_attempts=3, delay=1.0):
    """带参数的装饰器需要三层嵌套"""
    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            for attempt in range(1, max_attempts + 1):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    if attempt == max_attempts:
                        raise
                    print(f"第 {attempt} 次失败: {e}，{delay}s 后重试...")
                    time.sleep(delay)
        return wrapper
    return decorator

@retry(max_attempts=3, delay=0.5)
def call_unstable_api(prompt: str) -> str:
    import random
    if random.random() < 0.7:  # 70% 概率失败
        raise ConnectionError("API 超时")
    return f"Response to: {prompt}"

# 调用时自动重试
result = call_unstable_api("hello")
```

### 3.3 类装饰器（更优雅的带参数方案）

```python
class retry:
    def __init__(self, max_attempts=3, delay=1.0):
        self.max_attempts = max_attempts
        self.delay = delay

    def __call__(self, func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            for attempt in range(1, self.max_attempts + 1):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    if attempt == self.max_attempts:
                        raise
                    time.sleep(self.delay)
        return wrapper

@retry(max_attempts=5, delay=2.0)  # 更直观
def call_llm(prompt): ...
```

## 4. 高级用法

### 4.1 叠加多个装饰器（执行顺序）

```python
# 执行顺序：从下到上应用，从上到下执行
@timer        # 第2层包装
@retry(3)     # 第1层包装（先应用）
def fetch_data():
    ...

# 等价于：timer(retry(3)(fetch_data))
# 调用时：timer的wrapper → retry的wrapper → fetch_data
```

### 4.2 装饰器工厂 + 运行时状态

```python
def rate_limiter(calls_per_second: float):
    """限流装饰器：控制函数调用频率"""
    min_interval = 1.0 / calls_per_second
    last_called = [0.0]  # 用列表存储可变状态（闭包技巧）

    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            elapsed = time.time() - last_called[0]
            wait_time = min_interval - elapsed
            if wait_time > 0:
                time.sleep(wait_time)
            last_called[0] = time.time()
            return func(*args, **kwargs)
        return wrapper
    return decorator

@rate_limiter(calls_per_second=2)  # 每秒最多2次
def call_openai_api(prompt: str) -> str:
    return f"response: {prompt}"
```

## 5. AI Agent 中的实际应用

```python
import functools
import logging
import time
from typing import Callable, Any

logger = logging.getLogger(__name__)

# ---- 装饰器工具库（Agent 工程常用） ----

def llm_tool(description: str):
    """
    将函数注册为 LLM 可调用的工具
    类似 LangChain 的 @tool 装饰器原理
    """
    def decorator(func: Callable) -> Callable:
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            logger.info(f"Tool called: {func.__name__}, args={args}, kwargs={kwargs}")
            try:
                result = func(*args, **kwargs)
                logger.info(f"Tool result: {result}")
                return result
            except Exception as e:
                logger.error(f"Tool error: {e}")
                return f"Error: {str(e)}"  # LLM 友好的错误信息

        # 把元数据挂载到函数上，供 Agent 框架读取
        wrapper.tool_description = description
        wrapper.is_llm_tool = True
        return wrapper
    return decorator


def cache_response(ttl_seconds: int = 300):
    """LLM 响应缓存（相同 prompt 不重复调用）"""
    cache = {}

    def decorator(func: Callable) -> Callable:
        @functools.wraps(func)
        def wrapper(prompt: str, **kwargs) -> str:
            cache_key = f"{func.__name__}:{prompt}"
            now = time.time()

            if cache_key in cache:
                result, timestamp = cache[cache_key]
                if now - timestamp < ttl_seconds:
                    logger.debug(f"Cache hit: {cache_key[:50]}")
                    return result

            result = func(prompt, **kwargs)
            cache[cache_key] = (result, now)
            return result
        return wrapper
    return decorator


# ---- 实际使用 ----

@llm_tool(description="Search the web for current information")
@retry(max_attempts=3, delay=1.0)
@rate_limiter(calls_per_second=1)
def web_search(query: str) -> str:
    """真实的搜索工具函数"""
    # 实际会调用 SerpAPI / Tavily 等
    return f"Search results for: {query}"


@llm_tool(description="Execute Python code and return output")
def python_repl(code: str) -> str:
    """代码执行工具"""
    import io
    from contextlib import redirect_stdout
    output = io.StringIO()
    with redirect_stdout(output):
        exec(code)
    return output.getvalue()


# Agent 可以动态发现所有工具
def get_all_tools(module) -> list:
    import inspect
    tools = []
    for name, obj in inspect.getmembers(module, inspect.isfunction):
        if getattr(obj, 'is_llm_tool', False):
            tools.append({
                "name": name,
                "description": obj.tool_description,
                "function": obj
            })
    return tools
```

## 6. 常见错误

```python
# ❌ 错误1：忘记 functools.wraps，破坏元信息
def bad_decorator(func):
    def wrapper(*args, **kwargs):  # 没有 @functools.wraps
        return func(*args, **kwargs)
    return wrapper

@bad_decorator
def my_func():
    """重要的文档"""
    pass

print(my_func.__name__)  # 输出 "wrapper" 而不是 "my_func"
print(my_func.__doc__)   # 输出 None！

# ✅ 正确
def good_decorator(func):
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        return func(*args, **kwargs)
    return wrapper

# ❌ 错误2：带参数装饰器层数写错
def wrong_retry(max_attempts):  # 少了一层
    @functools.wraps(max_attempts)  # max_attempts 是 int！会报错
    def wrapper(*args, **kwargs):
        ...
    return wrapper

# ❌ 错误3：装饰器修改了函数签名，破坏类型检查
# 解决方案：使用 ParamSpec（Python 3.10+）
from typing import ParamSpec, TypeVar
P = ParamSpec('P')
R = TypeVar('R')

def typed_decorator(func: Callable[P, R]) -> Callable[P, R]:
    @functools.wraps(func)
    def wrapper(*args: P.args, **kwargs: P.kwargs) -> R:
        return func(*args, **kwargs)
    return wrapper
```

# 二、上下文管理器（Context Manager）

## 1. 为什么存在

**核心问题：** 如何保证资源在使用后**一定被释放**，即使发生异常？

```python
# 没有上下文管理器：
file = open("data.txt")
data = file.read()  # 如果这里抛异常 →
file.close()        # 这行永远不会执行！文件泄漏！

# 有了上下文管理器：
with open("data.txt") as file:
    data = file.read()  # 无论是否异常，文件都会被关闭
```

适用场景：文件、数据库连接、网络连接、锁、临时状态变更、LLM 会话管理...

---

## 2. Python 底层原理

上下文管理器协议只需两个魔术方法：

```python
class MyContext:
    def __enter__(self):
        """进入 with 块时调用，返回值赋给 as 变量"""
        print("进入上下文")
        return self  # 返回给 "as" 后的变量

    def __exit__(self, exc_type, exc_val, exc_tb):
        """
        退出 with 块时调用（无论是否异常）
        exc_type: 异常类型（无异常则为 None）
        exc_val:  异常值
        exc_tb:   traceback 对象
        返回 True：吞掉异常；返回 False/None：异常继续传播
        """
        print(f"退出上下文，异常: {exc_type}")
        return False  # 不吞掉异常

with MyContext() as ctx:
    print("在上下文内")
    # raise ValueError("test")  # 即使抛异常，__exit__ 也会被调用
```

---

## 3. 语法和核心用法

### 3.1 类方式实现

```python
import time

class Timer:
    """计时上下文管理器"""

    def __enter__(self):
        self.start = time.perf_counter()
        return self  # 返回 self，让外部可以访问 elapsed

    def __exit__(self, *args):
        self.elapsed = time.perf_counter() - self.start
        print(f"耗时: {self.elapsed:.4f}s")
        return False

with Timer() as t:
    time.sleep(0.1)
    # 做一些事情

print(f"精确耗时: {t.elapsed:.4f}s")  # 通过 as 变量访问
```

### 3.2 `contextlib.contextmanager`（推荐！更简洁）

```python
from contextlib import contextmanager

@contextmanager
def managed_connection(host: str):
    """数据库连接管理"""
    print(f"建立连接: {host}")
    conn = {"host": host, "status": "open"}  # 模拟连接对象

    try:
        yield conn          # yield 前 = __enter__
                            # yield 的值 = as 后的变量
    except Exception as e:
        print(f"发生异常，回滚: {e}")
        raise               # 重新抛出
    finally:
        conn["status"] = "closed"
        print(f"关闭连接: {host}")  # yield 后 = __exit__

with managed_connection("localhost:5432") as conn:
    print(f"使用连接: {conn}")
    # 模拟操作
```

> **`yield` 的魔法：**
>
> - `yield` **之前**的代码 = `__enter__`
> - `yield` **的值** = `as` 变量
> - `yield` **之后**的代码（finally）= `__exit__`

### 4.1 嵌套上下文管理器

```python
with (
```

### 4.2 异常处理策略

```python
@contextmanager
def suppress_and_log(*exception_types):
    """选择性吞掉指定异常（类似 contextlib.suppress）"""
    try:
        yield
    except exception_types as e:
        logger.warning(f"Suppressed: {type(e).__name__}: {e}")
        # 不 raise → 异常被吞掉

with suppress_and_log(ConnectionError, TimeoutError):
    call_unstable_api("test")  # 失败也不会崩溃
```

### 4.3 上下文管理器 + 异步（AI Agent 必备）

```python
from contextlib import asynccontextmanager
import asyncio

@asynccontextmanager
async def async_llm_session(model: str):
    """异步 LLM 会话管理"""
    print(f"初始化模型: {model}")
    session = {"model": model, "tokens_used": 0}

    try:
        yield session
    finally:
        print(f"会话结束，共使用 {session['tokens_used']} tokens")

async def main():
    async with async_llm_session("gpt-4") as session:
        # 模拟 LLM 调用
        session["tokens_used"] += 500
        await asyncio.sleep(0.1)

asyncio.run(main())
```

## 5. AI Agent 中的实际应用

```python
from contextlib import contextmanager
from typing import Generator
import threading

# ---- Agent 执行上下文 ----

@contextmanager
def agent_execution_context(
    agent_name: str,
    max_tokens: int = 4096,
    timeout: int = 30
) -> Generator[dict, None, None]:
    """
    Agent 执行的完整生命周期管理
    - 初始化执行状态
    - 设置超时
    - 记录 token 使用
    - 清理资源
    """
    context = {
        "agent": agent_name,
        "start_time": time.time(),
        "tokens_used": 0,
        "tool_calls": [],
        "max_tokens": max_tokens,
    }

    logger.info(f"Agent [{agent_name}] 开始执行")

    try:
        yield context

    except TimeoutError:
        logger.error(f"Agent [{agent_name}] 执行超时")
        raise
    except Exception as e:
        logger.error(f"Agent [{agent_name}] 执行失败: {e}")
        context["error"] = str(e)
        raise
    finally:
        elapsed = time.time() - context["start_time"]
        logger.info(
            f"Agent [{agent_name}] 完成 | "
            f"耗时: {elapsed:.2f}s | "
            f"Tokens: {context['tokens_used']} | "
            f"工具调用: {len(context['tool_calls'])} 次"
        )


# ---- 使用示例 ----

def run_research_agent(question: str) -> str:
    with agent_execution_context("ResearchAgent", max_tokens=8192) as ctx:
        # 模拟 Agent 工作流
        ctx["tool_calls"].append("web_search")
        ctx["tokens_used"] += 1200

        ctx["tool_calls"].append("summarize")
        ctx["tokens_used"] += 800

        return f"Research result for: {question}"


# ---- LLM 流式输出上下文管理器 ----

@contextmanager
def streaming_response(show_progress: bool = True):
    """管理 LLM 流式输出的显示状态"""
    chunks = []
    print("🤖 ", end="", flush=True)

    try:
        yield chunks  # Agent 代码往 chunks 里 append
    finally:
        if show_progress:
            print()  # 换行
        full_response = "".join(chunks)
        logger.debug(f"完整响应长度: {len(full_response)} 字符")
```

## 6. 常见错误

```python
# ❌ 错误1：contextmanager 里有多个 yield
@contextmanager
def bad_context():
    yield "first"
    yield "second"  # RuntimeError: generator didn't stop

# ❌ 错误2：没有处理 finally，异常时资源泄漏
@contextmanager
def leaky_context():
    resource = acquire_resource()
    yield resource
    release_resource(resource)  # 异常时不会执行！

# ✅ 正确：始终用 try/finally
@contextmanager
def safe_context():
    resource = acquire_resource()
    try:
        yield resource
    finally:
        release_resource(resource)  # 一定执行

# ❌ 错误3：在 __exit__ 中盲目返回 True（吞掉所有异常）
def __exit__(self, exc_type, exc_val, exc_tb):
    self.cleanup()
    return True  # 危险！调试时看不到任何异常

# ✅ 只吞掉特定异常
def __exit__(self, exc_type, exc_val, exc_tb):
    self.cleanup()
    return exc_type is not None and issubclass(exc_type, ExpectedError)
```

# 三、类型注解（Type Hints）

## 1. 为什么存在

Python 是动态类型语言，但大型项目（尤其 AI 工程）需要：

- **IDE 自动补全和跳转**
- **静态分析工具（mypy/pyright）提前发现错误**
- **文档即代码（函数签名即文档）**
- **LLM 代码生成质量提升（AI 读类型注解理解意图）**

```python
# 没有类型注解：这函数接受什么？返回什么？
def process(data, config):
    ...

# 有类型注解：一目了然
def process(data: list[dict], config: AgentConfig) -> ProcessResult:
    ...
```

---

## 1. 为什么存在

**核心问题：** 如何优雅地定义数据结构，而不写一堆重复的 `__init__`、`__repr__`、`__eq__`？

```python

class UserConfig:

        self.name = name
        self.model = model
        self.temperature = temperature
        self.max_tokens = max_tokens

    def __repr__(self):
        return f"UserConfig(name={self.name!r}, ...)"  # 要手写！

    def __eq__(self, other):
        return (self.name == other.name and ...)  # 要手写！
```

## 2. dataclass

### 2.1 底层原理

`@dataclass` 是一个**类装饰器**，它在类定义时检查字段注解，然后**自动生成** `__init__`、`__repr__`、`__eq__` 等方法。

```python
from dataclasses import dataclass, field
import inspect

@dataclass
class Point:
    x: float
    y: float

# 查看自动生成的 __init__ 签名
print(inspect.signature(Point.__init__))
# (self, x: float, y: float) -> None
```

### 2.2 核心语法

```python
from dataclasses import dataclass, field
from typing import Optional

@dataclass
class AgentConfig:
    # 必填字段（无默认值）
    name: str
    model: str

    # 有默认值的字段
    temperature: float = 0.7
    max_tokens: int = 4096

    # 可变默认值必须用 field(default_factory=...)
    # ❌ tools: list = []  # 所有实例共享同一个列表！
    tools: list[str] = field(default_factory=list)
    metadata: dict = field(default_factory=dict)

    # 不参与 __init__ 的字段
    _call_count: int = field(default=0, init=False, repr=False)

    # 计算字段
    def __post_init__(self):
        """__init__ 后自动调用，用于验证和计算"""
        if not (0 <= self.temperature <= 2):
            raise ValueError(f"temperature 必须在 0-2 之间，得到 {self.temperature}")
        if self.name:
            self.name = self.name.strip()


config = AgentConfig(name="ResearchAgent", model="gpt-4", temperature=0.3)
print(config)
# AgentConfig(name='ResearchAgent', model='gpt-4', temperature=0.3, ...)

# 自动 __eq__
config2 = AgentConfig(name="ResearchAgent", model="gpt-4", temperature=0.3)
print(config == config2)  # True！不需要手写

# frozen=True：不可变对象（可以作为 dict key 或 set 元素）

class ModelKey:
    provider: str
    model_name: str

```

## 3. Pydantic

Pydantic 是 `dataclass` 的"加强版"，增加了：

- **运行时数据验证**（最大特点）
- **类型强制转换**
- **JSON 序列化/反序列化**
- **从环境变量、文件加载配置**

### 3.1 核心语法

```python
from pydantic import BaseModel, Field, validator, model_validator
from typing import Literal, Optional

class AgentConfig(BaseModel):
    name: str
    model: str = "gpt-4"
    temperature: float = Field(default=0.7, ge=0, le=2)  # ge=大于等于, le=小于等于
    max_tokens: int = Field(default=4096, gt=0, le=32768)
    tools: list[str] = Field(default_factory=list)
    system_prompt: Optional[str] = None

    # 字段级验证器
    @validator('name')
    def name_not_empty(cls, v):
        if not v.strip():
            raise ValueError("name 不能为空字符串")
        return v.strip()

    # 模型级验证器（可以访问多个字段）
    @model_validator(mode='after')
    def check_consistency(self):
        if 'code_executor' in self.tools and self.model == 'gpt-3.5-turbo':
            raise ValueError("code_executor 工具需要 gpt-4 或更强的模型")
        return self


# ---- 运行时验证 ----
try:
    config = AgentConfig(
        name="  ResearchAgent  ",
        temperature=3.0  # 超出范围！
    )
except Exception as e:
    print(e)
    # 1 validation error for AgentConfig
    # temperature: Input should be less than or equal to 2

# ---- 类型强制转换 ----
config = AgentConfig(
    name="Agent",
    temperature="0.5",  # 字符串！Pydantic 自动转换为 float
    max_tokens="1000",  # 字符串→int
)
print(type(config.temperature))  # <class 'float'>
```

# 五、完整项目：Mini AI Agent 框架

整合以上四个技术点：

```python
"""
Mini AI Agent Framework
整合：装饰器 + 上下文管理器 + 类型注解 + Pydantic
"""

import time
import logging
import functools
from contextlib import contextmanager
from typing import Callable, Any, Generator
from pydantic import BaseModel, Field, validator
from dataclasses import dataclass, field

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)


# ==================== 类型定义（Pydantic）====================

class ToolDefinition(BaseModel):
    """工具的元数据定义"""
    name: str
    description: str
    parameters: dict[str, str] = {}

class AgentConfig(BaseModel):
    """Agent 配置，含运行时验证"""
    name: str
    model: str = "gpt-4"
    temperature: float = Field(0.7, ge=0, le=2)
    max_iterations: int = Field(10, gt=0, le=50)
    tools: list[str] = []

    @validator('name')
    def validate_name(cls, v):
        return v.strip() or (_ for _ in ()).throw(ValueError("name 不能为空"))

class AgentMessage(BaseModel):
    """对话消息"""
    role: str
    content: str
    metadata: dict[str, Any] = {}

class AgentResult(BaseModel):
    """Agent 执行结果"""
    success: bool
    output: str
    iterations: int
    tokens_used: int
    tool_calls: list[str] = []
    error: str | None = None


# ==================== 装饰器库 ====================

def tool(description: str):
    """注册函数为 Agent 工具"""
    def decorator(func: Callable) -> Callable:
        @functools.wraps(func)
        def wrapper(*args, **kwargs) -> str:
            logger.debug(f"Tool [{func.__name__}] called: args={args}")
            try:
                result = func(*args, **kwargs)
                return str(result)
            except Exception as e:
                return f"[Tool Error] {func.__name__}: {e}"

        wrapper._is_tool = True
        wrapper._tool_def = ToolDefinition(
            name=func.__name__,
            description=description,
        )
        return wrapper
    return decorator


def retry(max_attempts: int = 3, delay: float = 1.0, exceptions: tuple = (Exception,)):
    """智能重试装饰器"""
    def decorator(func: Callable) -> Callable:
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            for attempt in range(1, max_attempts + 1):
                try:
                    return func(*args, **kwargs)
                except exceptions as e:
                    if attempt == max_attempts:
                        logger.error(f"[{func.__name__}] 重试 {max_attempts} 次后失败")
                        raise
                    logger.warning(f"[{func.__name__}] 第{attempt}次失败: {e}")
                    time.sleep(delay * attempt)  # 指数退避
        return wrapper
    return decorator


def log_execution(func: Callable) -> Callable:
    """执行日志装饰器"""
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        logger.info(f"▶ {func.__name__} 开始")
        start = time.perf_counter()
        try:
            result = func(*args, **kwargs)
            elapsed = time.perf_counter() - start
            logger.info(f"✓ {func.__name__} 完成 ({elapsed:.3f}s)")
            return result
        except Exception as e:
            elapsed = time.perf_counter() - start
            logger.error(f"✗ {func.__name__} 失败 ({elapsed:.3f}s): {e}")
            raise
    return wrapper


# ==================== 上下文管理器 ====================

@contextmanager
def agent_run_context(
    config: AgentConfig,
) -> Generator[dict, None, None]:
    """Agent 完整运行上下文"""
    state = {
        "config": config,
        "start_time": time.time(),
        "tokens_used": 0,
        "tool_calls": [],
        "iteration": 0,
    }

    logger.info(f"🤖 Agent [{config.name}] 启动 | 模型: {config.model}")

    try:
        yield state
    except Exception as e:
        state["error"] = str(e)
        logger.error(f"❌ Agent [{config.name}] 运行异常: {e}")
        raise
    finally:
        elapsed = time.time() - state["start_time"]
        logger.info(
            f"🏁 Agent [{config.name}] 结束 | "
            f"耗时: {elapsed:.2f}s | "
            f"迭代: {state['iteration']} | "
            f"Tools: {len(state['tool_calls'])} 次"
        )


# ==================== 工具集 ====================

@tool(description="搜索网络获取最新信息")
@retry(max_attempts=2, delay=0.5)
def web_search(query: str) -> str:
    """模拟网络搜索"""
    time.sleep(0.1)  # 模拟网络延迟
    return f"[搜索结果] '{query}' 的相关信息：这是模拟的搜索结果。"


@tool(description="执行 Python 代码并返回输出")
def python_repl(code: str) -> str:
    """安全的代码执行环境"""
    import io
    from contextlib import redirect_stdout
    output = io.StringIO()
    try:
        with redirect_stdout(output):
            exec(code, {"__builtins__": {"print": print, "range": range, "len": len}})
        return output.getvalue() or "[无输出]"
    except Exception as e:
        return f"[执行错误] {e}"


@tool(description="计算数学表达式")
def calculator(expression: str) -> str:
    """安全计算器"""
    try:
        # 简单安全检查
        allowed = set("0123456789+-*/()., ")
        if not all(c in allowed for c in expression):
            return "[错误] 包含非法字符"
        result = eval(expression)
        return f"{expression} = {result}"
    except Exception as e:
        return f"[计算错误] {e}"


# ==================== Agent 核心 ====================

@dataclass
class ToolRegistry:
    """工具注册表"""
    _tools: dict[str, Callable] = field(default_factory=dict)

    def register(self, func: Callable) -> None:
        if getattr(func, '_is_tool', False):
            self._tools[func.__name__] = func

    def get(self, name: str) -> Callable | None:
        return self._tools.get(name)

    def list_tools(self) -> list[ToolDefinition]:
        return [f._tool_def for f in self._tools.values()]

    def __repr__(self) -> str:
        return f"ToolRegistry(tools={list(self._tools.keys())})"


class MiniAgent:
    """
    Mini AI Agent 实现
    整合所有四个技术点的完整示例
    """

    def __init__(self, config: AgentConfig):
        self.config = config
        self.registry = ToolRegistry()
        self.history: list[AgentMessage] = []

        # 自动注册所有工具
        for func in [web_search, python_repl, calculator]:
            self.registry.register(func)

        logger.info(f"Agent [{config.name}] 初始化完成: {self.registry}")

    @log_execution
    def run(self, task: str) -> AgentResult:
        """Agent 主入口"""
        with agent_run_context(self.config) as state:
            self.history.append(AgentMessage(role="user", content=task))

            # 模拟 Agent ReAct 循环
            current_task = task
            final_answer = ""

            while state["iteration"] < self.config.max_iterations:
                state["iteration"] += 1

                # 模拟 LLM 决策（实际会调用真实 LLM API）
                action = self._simulate_llm_decision(current_task, state["iteration"])
                state["tokens_used"] += 200  # 模拟 token 消耗

                if action["type"] == "final":
                    final_answer = action["content"]
                    break

                elif action["type"] == "tool":
                    tool_name = action["tool"]
                    tool_args = action["args"]

                    tool_func = self.registry.get(tool_name)
                    if tool_func:
                        result = tool_func(**tool_args)
                        state["tool_calls"].append(tool_name)
                        state["tokens_used"] += 100

                        self.history.append(AgentMessage(
                            role="assistant",
                            content=f"[Tool: {tool_name}]\n输入: {tool_args}\n输出: {result}",
                            metadata={"tool": tool_name}
                        ))
                        current_task = f"基于工具结果回答原始问题: {task}\n工具结果: {result}"

            return AgentResult(
                success=bool(final_answer),
                output=final_answer or "达到最大迭代次数，未能完成任务",
                iterations=state["iteration"],
                tokens_used=state["tokens_used"],
                tool_calls=state["tool_calls"],
            )

    def _simulate_llm_decision(self, task: str, iteration: int) -> dict:
        """模拟 LLM 决策（实际应调用真实 API）"""
        # 简单的规则模拟，实际是 LLM 输出
        if iteration == 1 and ("搜索" in task or "查找" in task or "最新" in task):
            return {"type": "tool", "tool": "web_search", "args": {"query": task}}
        elif iteration == 1 and ("计算" in task or "等于" in task or any(c.isdigit() for c in task)):
            # 提取简单表达式
            import re
            expr_match = re.search(r'[\d\+\-\*\/\(\)\. ]+', task)
            expr = expr_match.group().strip() if expr_match else "1+1"
            return {"type": "tool", "tool": "calculator", "args": {"expression": expr}}
        else:
            return {"type": "final", "content": f"基于分析，任务 '{task}' 的答案是：[模拟的最终回答]"}


# ==================== 使用示例 ====================

if __name__ == "__main__":
    # 1. 创建配置（Pydantic 验证）
    try:
        config = AgentConfig(
            name="ResearchAgent",
            model="gpt-4",
            temperature=0.3,
            max_iterations=5,
            tools=["web_search", "calculator"],
        )
    except Exception as e:
        print(f"配置错误: {e}")
        exit(1)

    # 2. 创建 Agent
    agent = MiniAgent(config)

    # 3. 运行任务

        "解释什么是强化学习",
    ]

    for task in test_tasks:
        print(f"\n{'='*60}")
        print(f"任务: {task}")
        print('='*60)

        result = agent.run(task)

        # 结果也是 Pydantic 模型，可以直接序列化
        print(f"\n📊 执行结果:")
        print(result.model_dump_json(indent=2))
```

# 总结

| 技术                   | 核心价值              | AI Agent 场景                    |
| ---------------------- | --------------------- | -------------------------------- |
| **装饰器**             | 横切关注点分离        | 重试、限流、工具注册、日志       |
| **上下文管理器**       | 资源生命周期保证      | Agent 会话、连接池、执行状态     |
| **类型注解**           | 代码可读性 + 静态分析 | API 契约、IDE 支持、LLM 生成质量 |
| **dataclass/Pydantic** | 数据结构 + 验证       | 消息格式、配置管理、API 数据     |

这四个技术组合在一起，构成了现代 Python AI Agent 工程的**基础设施层**。
